git status - показывает какие файлы ты трогал и где ты сейчас находишься
git switch -c {имя} - переводит тебя в новую ветку с именем (-c - create)
git add  - добавляет в ветку файлы
git commit -m "текст(полезный)" - делает файлы частью твоей ветки ( позуй, но нужно писать сообщения и делить файлы) (-m - message)
git push -u origin {имя} - просишь сервер принять твои файлы в main (-u - union(союз))
git switch main - переводит тебя в main
git pull - получаешь последнюю версию main
